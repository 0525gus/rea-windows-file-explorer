cmake_minimum_required(VERSION 3.24)
project(REAExplorer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Concurrent Test)
add_library(explorer_core src/core/domain.cpp src/core/session.cpp include/explorer/session.hpp)
target_include_directories(explorer_core PUBLIC include)
target_link_libraries(explorer_core PUBLIC Qt6::Core Qt6::Gui Qt6::Concurrent)
if(WIN32)
  add_library(explorer_platform src/platform/windows/platform.cpp src/platform/windows/shell_services.cpp)
  target_link_libraries(explorer_platform PRIVATE ole32 shell32 shlwapi propsys uuid comctl32 gdi32 user32)
  target_compile_definitions(explorer_platform PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  add_library(explorer_platform src/platform/posix/platform.cpp)
endif()
target_sources(explorer_platform PRIVATE src/platform/common/file_operations.cpp)
target_include_directories(explorer_platform PUBLIC include)
target_link_libraries(explorer_platform PUBLIC Qt6::Core Qt6::Gui)
add_library(explorer_ui src/ui/main_window.cpp include/explorer/main_window.hpp)
target_include_directories(explorer_ui PUBLIC include)
target_link_libraries(explorer_ui PUBLIC explorer_core Qt6::Widgets)
add_executable(rea_explorer src/main.cpp)
if(WIN32)
  set_target_properties(rea_explorer PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
target_link_libraries(rea_explorer PRIVATE explorer_ui explorer_platform)
if(WIN32)
  add_custom_command(TARGET rea_explorer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:rea_explorer> $<TARGET_FILE_DIR:rea_explorer>
    COMMAND_EXPAND_LISTS)
endif()
if(MSVC)
  foreach(target explorer_core explorer_platform explorer_ui rea_explorer)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /Zc:__cplusplus)
  endforeach()
else()
  foreach(target explorer_core explorer_platform explorer_ui rea_explorer)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
  foreach(suite domain session platform ui operation)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${suite}_tests.cpp")
      add_executable(${suite}_tests "tests/${suite}_tests.cpp")
      target_link_libraries(${suite}_tests PRIVATE explorer_core explorer_platform explorer_ui Qt6::Test)
      if(WIN32)
        add_custom_command(TARGET ${suite}_tests POST_BUILD
          COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:${suite}_tests> $<TARGET_FILE_DIR:${suite}_tests>
          COMMAND_EXPAND_LISTS)
      endif()
      target_compile_definitions(${suite}_tests PRIVATE EXPLORER_PROJECT_DIR="${CMAKE_CURRENT_SOURCE_DIR}" EXPLORER_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
      if(MSVC)
        target_compile_options(${suite}_tests PRIVATE /utf-8)
      endif()
      add_test(NAME ${suite} COMMAND ${suite}_tests -o "${CMAKE_CURRENT_BINARY_DIR}/${suite}-results.txt,txt" -o "${CMAKE_CURRENT_BINARY_DIR}/${suite}-results.xml,junitxml")
      set_tests_properties(${suite} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
  endforeach()
endif()
install(TARGETS rea_explorer RUNTIME DESTINATION .)
