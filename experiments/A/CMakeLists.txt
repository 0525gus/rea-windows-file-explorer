cmake_minimum_required(VERSION 3.21)
project(ExplorerA VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Concurrent Test)
add_library(explorer_core
    src/core/navigation.cpp src/platform/filesystem_platform.cpp src/platform/file_operations.cpp
    src/application/browser_controller.cpp src/application/operation_manager.cpp
    include/explorer/browser_controller.h include/explorer/operation_manager.h)
target_include_directories(explorer_core PUBLIC include)
target_link_libraries(explorer_core PUBLIC Qt6::Core Qt6::Gui Qt6::Concurrent)
if(WIN32)
    target_link_libraries(explorer_core PRIVATE shell32)
endif()
add_library(explorer_ui src/ui/file_model.cpp src/ui/main_window.cpp src/ui/file_views.cpp
    include/explorer/file_model.h include/explorer/main_window.h include/explorer/file_views.h)
target_link_libraries(explorer_ui PUBLIC explorer_core Qt6::Widgets)
add_executable(explorer_a WIN32 src/main.cpp)
target_link_libraries(explorer_a PRIVATE explorer_ui)
add_executable(benchmark_a tests/benchmark.cpp)
target_link_libraries(benchmark_a PRIVATE explorer_ui)
foreach(target explorer_core explorer_ui explorer_a)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
enable_testing()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/test-reports")
set(EXPLORER_FIXTURE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../test-data/fixture-v1" CACHE PATH "Immutable synthetic fixture")
foreach(suite core platform ui operations management_ui)
    add_executable(test_${suite} tests/test_${suite}.cpp)
    target_link_libraries(test_${suite} PRIVATE explorer_ui Qt6::Test)
    target_compile_definitions(test_${suite} PRIVATE EXPLORER_FIXTURE_ROOT="${EXPLORER_FIXTURE_ROOT}")
    add_test(NAME ${suite} COMMAND test_${suite}
        -o "${CMAKE_BINARY_DIR}/test-reports/${suite}.xml,junitxml"
        -o "${CMAKE_BINARY_DIR}/test-reports/${suite}.txt,txt")
    set_tests_properties(${suite} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endforeach()
install(TARGETS explorer_a RUNTIME DESTINATION .)
