cmake_minimum_required(VERSION 3.24)
project(Phase6ComparisonHarness LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32)
  message(FATAL_ERROR "This validation harness imports the frozen A Windows/MSVC libraries.")
endif()
get_filename_component(COMPARISON_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(A_FROZEN_INCLUDE "${COMPARISON_PROJECT_ROOT}/experiments/A/artifacts/initial-freeze/include")
set(A_EXISTING_BUILD "${COMPARISON_PROJECT_ROOT}/experiments/A/build-windows")
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Widgets Concurrent)
set(A_EXPECTED_core "b557d6802cd79c061ec3ae4f763c8e79751ac27fc17c17805a18c700903ace5c")
set(A_EXPECTED_ui "83fa500d3891b8da4c2eba4464ae4032d9f7246ba657f9a9ed9fa2a21045375e")
foreach(component core ui)
  set(library "${A_EXISTING_BUILD}/explorer_${component}.lib")
  if(NOT EXISTS "${library}")
    message(FATAL_ERROR "Existing A library is missing: ${library}")
  endif()
  file(SHA256 "${library}" actual_digest)
  if(NOT actual_digest STREQUAL A_EXPECTED_${component})
    message(FATAL_ERROR "A imported library changed since harness preparation: ${library}")
  endif()
  add_library(a_initial_${component} STATIC IMPORTED)
  set_target_properties(a_initial_${component} PROPERTIES IMPORTED_LOCATION "${library}"
    INTERFACE_INCLUDE_DIRECTORIES "${A_FROZEN_INCLUDE}")
endforeach()
target_link_libraries(a_initial_core INTERFACE Qt6::Core Qt6::Gui Qt6::Concurrent shell32)
target_link_libraries(a_initial_ui INTERFACE a_initial_core Qt6::Widgets)
add_executable(comparison_a_gui a_gui_benchmark.cpp)
target_link_libraries(comparison_a_gui PRIVATE a_initial_ui)
target_compile_definitions(comparison_a_gui PRIVATE UNICODE _UNICODE)
if(MSVC)
  target_compile_options(comparison_a_gui PRIVATE /W4 /permissive- /utf-8 /Zc:__cplusplus)
endif()
add_custom_command(TARGET comparison_a_gui POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:comparison_a_gui> $<TARGET_FILE_DIR:comparison_a_gui>
  COMMAND_EXPAND_LISTS)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/a-runtime-dlls.txt"
  CONTENT "$<JOIN:$<TARGET_RUNTIME_DLLS:comparison_a_gui>,\n>\n")
